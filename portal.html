async function checkAuth() { try { // استخدام Supabase مباشرة (مثل باقي البوابات) if (!window.supabase || !window.AMIN_CONFIG) { throw new Error('Supabase not loaded'); } var sb = window.supabase.createClient( window.AMIN_CONFIG.supabaseUrl, window.AMIN_CONFIG.supabaseAnonKey, {auth:{persistSession:true, autoRefreshToken:true, storageKey:(window.AMIN_CONFIG.authStorageKey||'amin-ovcjzsrqqgjsbqswtkro-auth-v2')}} ); var sessionRes = await sb.auth.getSession(); var session = sessionRes.data.session; if (!session) { console.log('No session - redirecting to login'); location.href = 'index.html'; return null; } var userRes = await sb.from('users').select('*').eq('id', session.user.id).maybeSingle(); if (!userRes.data) { console.error('User not found in database'); location.href = 'index.html'; return null; } var user = userRes.data; var role = user.is_super_admin ? 'super_admin' : (user.role || 'guest'); if (!ROLE_SECTIONS[role]) { showToast('غير مصرّح', 'ليس لديك صلاحية الوصول لهذه البوابة', 'error'); setTimeout(function(){ location.href = 'index.html'; }, 2000); return null; } window.currentUser = user; window.currentUserRole = role; window.ME = user; // للتوافق مع smart-alerts console.log('✅ User authenticated:', user.name, '| Role:', role); return user; } catch (e) { console.error('Auth check failed:', e); setTimeout(function(){ location.href = 'index.html'; }, 1000); return null; } }
